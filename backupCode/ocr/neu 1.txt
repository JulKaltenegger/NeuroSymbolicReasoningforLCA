# ============================================================
# INDEX PARSING
# ============================================================

def clean_index_lines(text):
    lines = text.split("\n")
    cleaned = []

    for line in lines:
        line = line.strip()

        if not line:
            continue

        # remove dotted lines
        line = re.sub(r"\.{2,}", " ", line)

        cleaned.append(line)

    return cleaned


def normalize_page_number(page_str):
    page_str = page_str.replace("S", "5").replace("i", "1")
    return int(page_str)


def classify_index_lines(lines):
    sections = []
    titles = []
    pages = []

    for line in lines:

        # section numbers like 2 or 2.1
        if re.match(r"^\d+(\.\d+)?$", line):
            sections.append(line)

        # page numbers (small integers)
        elif re.match(r"^\d{1,3}$", line):
            pages.append(line)

        # everything else → title
        else:
            titles.append(line)

    return sections, titles, pages


def reconstruct_index(sections, titles, pages):
    entries = []

    n = min(len(sections), len(titles), len(pages))

    for i in range(n):
        entries.append({
            "section": sections[i],
            "title": titles[i],
            "page": normalize_page_number(pages[i])
        })

    return entries


def build_section_ranges(entries, last_page):
    sections = {}

    for i, entry in enumerate(entries):
        start = entry["page"]

        if i < len(entries) - 1:
            end = entries[i + 1]["page"] - 1
        else:
            end = last_page

        sections[entry["section"]] = {
            "title": entry["title"],
            "start": start,
            "end": end
        }

    return sections


# ============================================================
# PIPELINE EXECUTION
# ============================================================

lines = clean_index_lines(text)

print("\n--- PAGES ---\n", pages_list[:10])

entries = reconstruct_index(sections_list, titles_list, pages_list)

print("\n--- ENTRIES ---\n")
for e in entries:
    print(e)

# optional: use real page count later
sections = build_section_ranges(entries, last_page=100)

print("\n--- FINAL STRUCTURE ---\n")
for k, v in sections.items():
    print(k, ":", v)
