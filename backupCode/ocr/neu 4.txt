def group_rows(df, y_threshold=15):
    rows = []
    current_row = []
    current_y = None

    df_sorted = df.sort_values("top")

    for _, row in df_sorted.iterrows():
        if current_y is None:
            current_y = row["top"]

        if abs(row["top"] - current_y) < y_threshold:
            current_row.append(row)
        else:
            rows.append(current_row)
            current_row = [row]
            current_y = row["top"]

    if current_row:
        rows.append(current_row)

    return rows

def row_ends_with_page_number(row):
    if not row:
        return False
    last = normalize_ocr_token(row[-1].strip())
    return bool(re.fullmatch(r"\d+", last))


def is_section_header(row):
    """
    Banner-only parent line (e.g. category with no page column).
    Lines like "Blockbauweise 1,1 t 21" end with a page → handled as entries, not banners.
    """
    if not (len(row) >= 2 and not re.match(r"^\d", row[0])):
        return False
    if not any("bauweise" in w.lower() for w in row):
        return False
    if row_ends_with_page_number(row):
        return False
    return True


def normalize_ocr_token(token):
    """
    Normalize frequent OCR confusions for numeric fields.
    """
    return (
        token.replace("S", "5")
        .replace("s", "5")
        .replace("i", "1")
        .replace("I", "1")
        .replace("l", "1")
        .replace("|", "1")
        .replace("O", "0")
        .replace("o", "0")
    )


def normalize_section_token(token):
    """
    Normalize section numbering with safer logic.
    Exactly two digits with no dot are left as "35" for parse_section_from_row to reject,
    so extract_entries can treat them as M.N with missing dot and the correct chapter major.
    """
    cleaned = token.strip().replace(",", ".").replace(":", ".")
    cleaned = normalize_ocr_token(cleaned)

    # already correct
    if re.fullmatch(r"\d+\.\d+", cleaned):
        return cleaned

    # single number (e.g. "5") or long run (e.g. "100") — but not exactly two digits only
    if re.fullmatch(r"\d+", cleaned) and not re.fullmatch(r"\d{2}$", cleaned):
        return cleaned

    # "5U" after S→5 on "SU": dot between chapter and subsection is lost; "1" read as "U"
    m_u = re.fullmatch(r"(\d+)U", cleaned, re.I)
    if m_u:
        return f"{m_u.group(1)}.1"

    return cleaned


def parse_section_from_row(row_words):
    """
    Return normalized section token if row starts with one.
    Two digits only (e.g. "35") with no dot: OCR often dropped the dot in "5.5";
    return None so extract_entries maps major from the current Bauweise chapter.
    """
    if not row_words:
        return None

    candidate = normalize_section_token(row_words[0])
    if re.fullmatch(r"\d{2}$", candidate):
        return None
    if re.fullmatch(r"\d+(\.\d+)?", candidate):
        return candidate
    return None


def parse_page_from_row(row_words):
    """
    Return (page, title_tokens) for a row.
    Heuristic: page is usually right-most token and may be OCR-noisy.
    """
    if len(row_words) < 2:
        return None, row_words

    last = normalize_ocr_token(row_words[-1].strip())
    prev = row_words[-2].strip() if len(row_words) >= 3 else ""

    if re.fullmatch(r"\d+", last):
        page = int(last)

        # OCR-specific fix: "S. 5" is often read as "L. 4"
        if prev in {"L.", "L", "l.", "l"} and page == 4:
            page = 5

        return page, row_words[:-1]

    return None, row_words


def section_to_tuple(section):
    """
    Convert '2.1' -> (2, 1) for ordered comparisons.
    """
    if not section or not re.fullmatch(r"\d+(\.\d+)?", section):
        return None
    return tuple(int(p) for p in section.split("."))


def apply_toc_corrections(entries, corrections):
    """
    Merge optional corrections into extracted entries without document-specific defaults.

    ``corrections`` dict may contain:
      - ``by_entry_index``: { "3": {"section": "4.1", "title": "..."}, ... }
        Applies to merge order of entries (same order as extracted list).
      - ``by_ocr_row``: { "42": {"section": "3.1"}, ... }
        Applies to entries that carry "_ocr_row" matching structured_rows index.

    Only keys listed under ``set``/flat dict are patched onto the entry (shallow merge).
    """
    if not corrections:
        return entries

    by_idx = corrections.get("by_entry_index") or {}
    by_row = corrections.get("by_ocr_row") or {}

    def patch_entry(entry, patch):
        if not patch:
            return
        for k, v in patch.items():
            if v is None:
                entry.pop(k, None)
            else:
                entry[k] = v

    for i, e in enumerate(entries):
        key_str = str(i)
        if key_str in by_idx:
            patch_entry(e, by_idx[key_str])

    for e in entries:
        row_key = e.get("_ocr_row")
        if row_key is None:
            continue
        key_str = str(row_key)
        if key_str in by_row:
            patch_entry(e, by_row[key_str])

    return entries


def apply_toc_corrections_from_file(entries, path):
    if not path or not os.path.isfile(path):
        return entries
    with open(path, encoding="utf-8") as f:
        data = json.load(f)
    return apply_toc_corrections(entries, data)


def fill_missing_pages(entries):
    """
    Fill missing page numbers when neighboring section numbers are consecutive.
    """
    for i, entry in enumerate(entries):
        if entry.get("page") is not None:
            continue
        if i == 0:
            continue

        prev = entries[i - 1]
        prev_page = prev.get("page")
        if not isinstance(prev_page, int):
            continue

        curr_sec = section_to_tuple(entry.get("section"))
        prev_sec = section_to_tuple(prev.get("section"))
        if not curr_sec or not prev_sec:
            continue

        # Example: 2.1 -> 2.2, assign next page when OCR misses it.
        if len(curr_sec) == len(prev_sec) == 2 and curr_sec[0] == prev_sec[0] and curr_sec[1] == prev_sec[1] + 1:
            entry["page"] = prev_page + 1

    return entries

def parse_main_section_from_row(row_words):
    if not row_words:
        return None

    token = normalize_ocr_token(row_words[0])

    # ONLY pure integers allowed
    if re.fullmatch(r"\d+", token):
        return int(token)

    return None

def extract_entries(structured_rows):
    entries = []
    current_main_section = 1
    current_parent = None
    # Subsections remap to (current_main_section - 1) only AFTER we've seen a
    # Bauweise + page banner. Before that, remapping uses 1-1 == 0 → bogus "0.1";
    # the PDF OCR still has the real major (e.g. 2.1 → keep 2).
    saw_bauweise_main_banner = False
    # Highest M in M.N subsections encountered before first Bauweise+page banner.
    # TOC often has Einleitung + Blockbauweise 0,8 chapters without such a banner row,
    # so first visible banner ("Blockbauweise 1,1 t") must be chap. max+1 (e.g. 3).
    max_ocr_major_before_first_banner = 0

    for i, row in enumerate(structured_rows):

        if not row:
            continue

        # ============================================================
        # 1. MAIN SECTION (detect by "bauweise" + page)
        # ============================================================
        if (
            len(row) >= 2
            and not re.match(r"^\d", row[0])
            and any("bauweise" in w.lower() for w in row)
            and row_ends_with_page_number(row)
        ):
            page, title_tokens = parse_page_from_row(row)
            title = " ".join(title_tokens).strip()

            if not saw_bauweise_main_banner:
                current_main_section = max(
                    current_main_section,
                    max_ocr_major_before_first_banner + 1,
                )

            entries.append({
                "_ocr_row": i,
                "section": str(current_main_section),
                "title": title,
                "page": page,
                "parent": None
            })

            current_parent = title
            saw_bauweise_main_banner = True
            current_main_section += 1
            continue


        # ============================================================
        # 2. SUBSECTIONS
        # ============================================================
        section = parse_section_from_row(row)

        # OCR FIX: "35" → "5.5"
        if (
            not section
            and re.fullmatch(r"\d{2}", normalize_ocr_token(row[0]))
        ):
            token = normalize_ocr_token(row[0])
            sub = int(token[1])
            if saw_bauweise_main_banner:
                major = max(1, current_main_section - 1)
            else:
                major = int(token[0])
            section = f"{major}.{sub}"

        if section:

            # Map M.N to our running main chapter only after a banner set the counter;
            # otherwise keep OCR major (e.g. 2.1 before any Bauweise+page row).
            if "." in section:
                ocr_major_str, sub_str = section.split(".")
                ocr_major = int(ocr_major_str)
                sub = int(sub_str)
                if saw_bauweise_main_banner:
                    major = max(1, current_main_section - 1)
                    section = f"{major}.{sub}"
                else:
                    section = f"{ocr_major}.{sub}"

            page, title_tokens = parse_page_from_row(row[1:])
            title = " ".join(title_tokens).strip()

            # fallback page from next row
            if page is None and i + 1 < len(structured_rows):
                next_row = structured_rows[i + 1]
                if len(next_row) == 1:
                    token = normalize_ocr_token(next_row[0])
                    if re.fullmatch(r"\d+", token):
                        page = int(token)

            # enforce sequence
            if entries:
                prev = entries[-1]
                prev_sec = prev.get("section")

                if (
                    isinstance(prev_sec, str)
                    and "." in prev_sec
                    and "." in section
                ):
                    prev_main, prev_sub = map(int, prev_sec.split("."))
                    curr_main, curr_sub = map(int, section.split("."))

                    if curr_main == prev_main:
                        if curr_sub <= prev_sub:
                            curr_sub = prev_sub + 1
                            section = f"{curr_main}.{curr_sub}"

            if not saw_bauweise_main_banner and "." in section:
                m_maj = int(section.split(".")[0])
                max_ocr_major_before_first_banner = max(
                    max_ocr_major_before_first_banner, m_maj
                )

            entries.append({
                "_ocr_row": i,
                "section": section,
                "title": title,
                "page": page,
                "parent": current_parent
            })

            continue

    return fill_missing_pages(entries)

def row_to_json_record(row_words):
    """
    Build one JSON-friendly record from one OCR line.
    """
    line_text = " ".join(row_words).strip()
    record = {
        "line": line_text,
        "words": row_words,
        "section": None,
        "title": None,
        "page": None,
    }

    section = parse_section_from_row(row_words)
    if len(row_words) >= 2 and section:
        record["section"] = section
        page, title_tokens = parse_page_from_row(row_words[1:])
        record["page"] = page
        record["title"] = " ".join(title_tokens).strip() or None

    return record


def rows_to_json(structured_rows):
    records = [row_to_json_record(row) for row in structured_rows if row]

    # Mirror the same page-fill logic used for parsed entries.
    indexed_records = [r for r in records if r.get("section")]
    fill_missing_pages(indexed_records)

    return records

def format_entry_line(e):
    """Human-readable one line: '2.1 Typenserie … 5' or unnumbered category + page."""
    sec = e.get("section")
    title = (e.get("title") or "").strip()
    page = e.get("page")
    parts = []
    if sec:
        parts.append(str(sec))
    if title:
        parts.append(title)
    if page is not None:
        parts.append(str(page))
    return " ".join(parts)

def build_section_ranges(entries, last_page=100):
    sections = {}
    valid_entries = [
        e for e in entries
        if isinstance(e.get("page"), int) and e.get("section")
    ]

    for i, entry in enumerate(valid_entries):
        start = entry["page"]

        if i < len(valid_entries) - 1:
            end = valid_entries[i + 1]["page"] - 1
        else:
            end = last_page

        sections[entry["section"]] = {
            "title": entry["title"],
            "start": start,
            "end": end
        }

    return sections
